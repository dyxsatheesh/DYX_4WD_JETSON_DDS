cmake_minimum_required(VERSION 3.16)
project(dyx_mission_cpp17_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(dyx_rpp_cpp17_core REQUIRED)
add_executable(dyx_mission_cpp17_preview_node src/mission_preview_node.cpp)
target_include_directories(dyx_mission_cpp17_preview_node PRIVATE include)
ament_target_dependencies(dyx_mission_cpp17_preview_node
  rclcpp geometry_msgs std_msgs dyx_rpp_cpp17_core)
install(TARGETS dyx_mission_cpp17_preview_node DESTINATION lib/${PROJECT_NAME})
ament_package()
