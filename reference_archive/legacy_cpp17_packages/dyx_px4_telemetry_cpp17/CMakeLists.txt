cmake_minimum_required(VERSION 3.16)
project(dyx_px4_telemetry_cpp17 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(px4_msgs REQUIRED)
add_executable(dyx_px4_telemetry_node src/dyx_px4_telemetry_node.cpp)
target_include_directories(dyx_px4_telemetry_node PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/../dyx_mission_manager/include)
ament_target_dependencies(dyx_px4_telemetry_node rclcpp std_msgs px4_msgs)
install(TARGETS dyx_px4_telemetry_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
