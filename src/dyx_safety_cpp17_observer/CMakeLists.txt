cmake_minimum_required(VERSION 3.16)
project(dyx_safety_cpp17_observer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(dyx_rpp_cpp17_core REQUIRED)
add_executable(dyx_safety_cpp17_observer_node src/safety_observer_node.cpp)
target_include_directories(dyx_safety_cpp17_observer_node PRIVATE include)
ament_target_dependencies(dyx_safety_cpp17_observer_node
  rclcpp px4_msgs std_msgs dyx_rpp_cpp17_core)
install(TARGETS dyx_safety_cpp17_observer_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch)
ament_package()
