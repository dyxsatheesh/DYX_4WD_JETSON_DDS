cmake_minimum_required(VERSION 3.16)
project(dyx_jetson_readonly_dds LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
add_executable(dyx_jetson_readonly_dds_node src/readonly_dds_node.cpp)
target_include_directories(dyx_jetson_readonly_dds_node PRIVATE include)
ament_target_dependencies(dyx_jetson_readonly_dds_node rclcpp px4_msgs)
install(TARGETS dyx_jetson_readonly_dds_node DESTINATION lib/${PROJECT_NAME})
ament_package()
