cmake_minimum_required(VERSION 3.16)
project(rtk_correction_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(rtk_correction_bridge_node src/ntrip_observer_node.cpp)
target_include_directories(rtk_correction_bridge_node PRIVATE include)
ament_target_dependencies(rtk_correction_bridge_node rclcpp std_msgs)
target_link_libraries(rtk_correction_bridge_node OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
install(TARGETS rtk_correction_bridge_node DESTINATION lib/${PROJECT_NAME})
ament_package()
