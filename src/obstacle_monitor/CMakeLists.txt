cmake_minimum_required(VERSION 3.16)
project(obstacle_monitor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(obstacle_monitor_node src/obstacle_monitor_node.cpp)
ament_target_dependencies(obstacle_monitor_node rclcpp std_msgs sensor_msgs)
install(TARGETS obstacle_monitor_node DESTINATION lib/${PROJECT_NAME})
ament_package()
