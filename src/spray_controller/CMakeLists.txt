cmake_minimum_required(VERSION 3.16)
project(spray_controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(dyx_rpp_cpp17_core REQUIRED)
add_executable(spray_controller_node src/spray_controller_node.cpp)
ament_target_dependencies(spray_controller_node rclcpp std_msgs dyx_rpp_cpp17_core)
install(TARGETS spray_controller_node DESTINATION lib/${PROJECT_NAME})
ament_package()
