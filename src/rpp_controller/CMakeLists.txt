cmake_minimum_required(VERSION 3.16)
project(rpp_controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(dyx_rpp_cpp17_core REQUIRED)
find_package(px4_msgs REQUIRED)
add_executable(rpp_controller_node src/rpp_controller_node.cpp)
ament_target_dependencies(rpp_controller_node rclcpp std_msgs geometry_msgs nav_msgs dyx_rpp_cpp17_core px4_msgs)
install(TARGETS rpp_controller_node DESTINATION lib/${PROJECT_NAME})
ament_package()
