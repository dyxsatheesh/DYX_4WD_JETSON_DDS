cmake_minimum_required(VERSION 3.16)
project(jetson_4wd_control LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(cmd_vel_bridge src/cmd_vel_bridge.cpp)
target_include_directories(cmd_vel_bridge PRIVATE include)
ament_target_dependencies(cmd_vel_bridge rclcpp std_msgs)
install(TARGETS cmd_vel_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
