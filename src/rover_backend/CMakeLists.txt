cmake_minimum_required(VERSION 3.16)
project(rover_backend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(dyx_rpp_cpp17_core REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(Drogon CONFIG REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
add_executable(rover_backend src/rover_backend.cpp)
ament_target_dependencies(rover_backend rclcpp std_msgs dyx_rpp_cpp17_core px4_msgs)
target_include_directories(rover_backend PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../dyx_mission_manager/include)
target_link_libraries(rover_backend Drogon::Drogon Threads::Threads OpenSSL::Crypto sqlite3)
install(TARGETS rover_backend DESTINATION lib/${PROJECT_NAME})
add_executable(rtk_worker src/rtk_worker.cpp)
ament_target_dependencies(rtk_worker rclcpp std_msgs dyx_rpp_cpp17_core px4_msgs)
target_include_directories(rtk_worker PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../dyx_mission_manager/include)
target_link_libraries(rtk_worker Drogon::Drogon Threads::Threads OpenSSL::Crypto sqlite3)
install(TARGETS rtk_worker DESTINATION lib/${PROJECT_NAME})
add_executable(estimator_status_relay src/estimator_status_relay.cpp)
ament_target_dependencies(estimator_status_relay rclcpp std_msgs dyx_rpp_cpp17_core px4_msgs)
target_include_directories(estimator_status_relay PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../dyx_mission_manager/include)
target_link_libraries(estimator_status_relay Drogon::Drogon Threads::Threads OpenSSL::Crypto sqlite3)
install(TARGETS estimator_status_relay DESTINATION lib/${PROJECT_NAME})
ament_package()
