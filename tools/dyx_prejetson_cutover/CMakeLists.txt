cmake_minimum_required(VERSION 3.16)
project(dyx_prejetson_cutover LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(dyx_cutover_audit_lib INTERFACE)
target_include_directories(dyx_cutover_audit_lib INTERFACE include)
add_executable(dyx_cutover_audit src/audit.cpp)
target_link_libraries(dyx_cutover_audit PRIVATE dyx_cutover_audit_lib)
add_executable(cutover_guard_tests test/guard_tests.cpp)
target_link_libraries(cutover_guard_tests PRIVATE dyx_cutover_audit_lib)
if(MSVC)
  target_compile_options(dyx_cutover_audit PRIVATE /W4)
  target_compile_options(cutover_guard_tests PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(dyx_cutover_audit PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cutover_guard_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
add_test(NAME cutover_offline_synthetic_regression COMMAND cutover_guard_tests)
