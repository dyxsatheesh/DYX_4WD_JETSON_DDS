cmake_minimum_required(VERSION 3.16)
project(dyx_h6d_gate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
set(H6D_CORE_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../src/dyx_rpp_cpp17_core/include")
set(H6D_OBSERVER_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../src/dyx_safety_cpp17_observer/include")
add_executable(dyx_h6d_readonly_test test/readonly_test.cpp)
target_include_directories(dyx_h6d_readonly_test PRIVATE "${H6D_OBSERVER_INCLUDE}")
add_executable(dyx_h6d_safety_core_test test/safety_core_test.cpp)
target_include_directories(dyx_h6d_safety_core_test PRIVATE "${H6D_CORE_INCLUDE}")
add_executable(dyx_h6d_source_gate test/source_gate.cpp)
foreach(t IN ITEMS dyx_h6d_readonly_test dyx_h6d_safety_core_test dyx_h6d_source_gate)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /UNDEBUG)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  endif()
endforeach()
add_test(NAME h6d_read_only_contract COMMAND dyx_h6d_readonly_test)
add_test(NAME h6d_existing_safety_core COMMAND dyx_h6d_safety_core_test)
add_test(NAME h6d_source_no_actuation COMMAND dyx_h6d_source_gate
 "${CMAKE_CURRENT_SOURCE_DIR}/../../src/dyx_safety_cpp17_observer/src/safety_observer_node.cpp")
