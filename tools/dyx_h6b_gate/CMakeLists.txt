cmake_minimum_required(VERSION 3.16)
project(dyx_h6b_gate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(dyx_h6b_gate_tests test/gate_tests.cpp)
target_include_directories(dyx_h6b_gate_tests PRIVATE
  ../../src/dyx_rpp_cpp17_native_preview/include)
if(MSVC)
  target_compile_options(dyx_h6b_gate_tests PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(dyx_h6b_gate_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
add_test(NAME h6b_preview_gate_safety COMMAND dyx_h6b_gate_tests)
add_executable(dyx_h6b_static_gate test/static_boundary.cpp)
if(MSVC)
  target_compile_options(dyx_h6b_static_gate PRIVATE /W4)
else()
  target_compile_options(dyx_h6b_static_gate PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME h6b_no_actuation_source_gate COMMAND dyx_h6b_static_gate
  "${CMAKE_CURRENT_SOURCE_DIR}/../../src/dyx_rpp_cpp17_native_preview/src/native_preview_node.cpp")
