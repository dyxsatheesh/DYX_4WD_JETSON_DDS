cmake_minimum_required(VERSION 3.16)
project(dyx_h6a_observer_gate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(dyx_h6a_lib INTERFACE)
target_include_directories(dyx_h6a_lib INTERFACE include)
add_executable(dyx_h6a_gate src/verify.cpp)
add_executable(dyx_h6a_gate_tests test/gate_tests.cpp)
target_link_libraries(dyx_h6a_gate PRIVATE dyx_h6a_lib)
target_link_libraries(dyx_h6a_gate_tests PRIVATE dyx_h6a_lib)
if(MSVC)
  target_compile_options(dyx_h6a_gate PRIVATE /W4)
  target_compile_options(dyx_h6a_gate_tests PRIVATE /W4)
else()
  target_compile_options(dyx_h6a_gate PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(dyx_h6a_gate_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME h6a_offline_mutation_guard COMMAND dyx_h6a_gate_tests)
