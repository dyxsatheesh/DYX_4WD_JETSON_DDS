cmake_minimum_required(VERSION 3.16)
project(dyx_h6e2_gate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
set(INC ${CMAKE_CURRENT_SOURCE_DIR}/../../src/dyx_backend_cpp17_legacy_readonly/include)
foreach(name IN ITEMS contract_test json_test source_gate)
  add_executable(dyx_h6e2_${name} test/${name}.cpp)
  target_include_directories(dyx_h6e2_${name} PRIVATE ${INC} test)
  if(MSVC)
    target_compile_options(dyx_h6e2_${name} PRIVATE /W4 /UNDEBUG)
  else()
    target_compile_options(dyx_h6e2_${name} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  endif()
endforeach()
add_test(NAME h6e2_legacy_readonly_route_contract COMMAND dyx_h6e2_contract_test)
add_test(NAME h6e2_strict_json_and_latching COMMAND dyx_h6e2_json_test)
add_test(NAME h6e2_no_actuation_and_legacy_source COMMAND dyx_h6e2_source_gate
  "${INC}/dyx_backend_cpp17_legacy_readonly/legacy_readonly.hpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/../..")
