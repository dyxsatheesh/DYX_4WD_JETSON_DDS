cmake_minimum_required(VERSION 3.16)
project(dyx_jetson_preflight LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_executable(dyx_jetson_preflight src/main.cpp)
target_include_directories(dyx_jetson_preflight PRIVATE include)
add_executable(preflight_inhibited test/inhibited.cpp)
target_include_directories(preflight_inhibited PRIVATE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(preflight_inhibited PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
endif()
add_test(NAME h5u_output_inhibited COMMAND preflight_inhibited)
add_test(NAME h5u_binary_inspection COMMAND dyx_jetson_preflight --check)

if(UNIX AND NOT APPLE)
  add_executable(preflight_notify test/notify.cpp)
  target_include_directories(preflight_notify PRIVATE include)
  target_compile_options(preflight_notify PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  add_test(NAME h5v_systemd_notify COMMAND preflight_notify)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(OpenSSL REQUIRED)
  find_package(SQLite3 REQUIRED)
  add_executable(dyx_jetson_unix_probe src/unix_probe.cpp)
  target_include_directories(dyx_jetson_unix_probe PRIVATE include)
  target_link_libraries(dyx_jetson_unix_probe PRIVATE OpenSSL::Crypto SQLite::SQLite3)
  target_compile_options(dyx_jetson_unix_probe PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(preflight_auth test/auth.cpp)
  target_include_directories(preflight_auth PRIVATE include)
  target_link_libraries(preflight_auth PRIVATE OpenSSL::Crypto)
  target_compile_options(preflight_auth PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  add_test(NAME h5w_hmac_auth COMMAND preflight_auth)
  add_executable(preflight_replay test/replay.cpp)
  target_include_directories(preflight_replay PRIVATE include)
  target_link_libraries(preflight_replay PRIVATE OpenSSL::Crypto SQLite::SQLite3)
  target_compile_options(preflight_replay PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  add_test(NAME h5x_persistent_replay COMMAND preflight_replay)
endif()

add_executable(preflight_readonly test/readonly.cpp)
target_include_directories(preflight_readonly PRIVATE
  ../../src/dyx_jetson_readonly_dds/include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(preflight_readonly PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
endif()
add_test(NAME h5y_dds_readonly_observation COMMAND preflight_readonly)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME h5y_ros_source_static_gate
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test/verify_ros_readonly.py")
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND Python3_Interpreter_FOUND)
  add_test(NAME h5z_real_unix_hmac_sqlite_restart
    COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/test/e2e_unix_probe.py"
      "$<TARGET_FILE:dyx_jetson_unix_probe>")
endif()
if(Python3_Interpreter_FOUND)
  add_test(NAME h5z_deployment_static_gate
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test/verify_deployment.py")
endif()
if(MSVC)
  foreach(unit IN ITEMS preflight_inhibited preflight_readonly)
    target_compile_options(${unit} PRIVATE /UNDEBUG /W4)
  endforeach()
endif()
