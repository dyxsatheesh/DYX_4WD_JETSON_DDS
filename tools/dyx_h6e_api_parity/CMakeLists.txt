cmake_minimum_required(VERSION 3.16)
project(dyx_h6e_api_parity LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_library(dyx_h6e_inventory INTERFACE)
target_include_directories(dyx_h6e_inventory INTERFACE include)
add_executable(dyx_h6e_api_audit src/audit.cpp)
target_link_libraries(dyx_h6e_api_audit PRIVATE dyx_h6e_inventory)
add_executable(dyx_h6e_parser_test test/parser_test.cpp)
target_link_libraries(dyx_h6e_parser_test PRIVATE dyx_h6e_inventory)
add_executable(dyx_h6e_repo_test test/repo_test.cpp)
target_link_libraries(dyx_h6e_repo_test PRIVATE dyx_h6e_inventory)
if(MSVC)
  foreach(t IN ITEMS dyx_h6e_api_audit dyx_h6e_parser_test dyx_h6e_repo_test)
    target_compile_options(${t} PRIVATE /W4 /UNDEBUG)
  endforeach()
else()
  foreach(t IN ITEMS dyx_h6e_api_audit dyx_h6e_parser_test dyx_h6e_repo_test)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  endforeach()
endif()
add_test(NAME h6e1_parser_contract COMMAND dyx_h6e_parser_test)
add_test(NAME h6e1_repo_contract COMMAND dyx_h6e_repo_test "${CMAKE_CURRENT_SOURCE_DIR}/../..")
