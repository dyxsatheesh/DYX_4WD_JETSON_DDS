cmake_minimum_required(VERSION 3.16)
project(dyx_h6c_gate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(dyx_h6c_selection_tests test/selection_tests.cpp)
target_include_directories(dyx_h6c_selection_tests PRIVATE ../../src/dyx_mission_cpp17_preview/include)
add_executable(dyx_h6c_static_gate test/static_gate.cpp)
if(MSVC)
  target_compile_options(dyx_h6c_selection_tests PRIVATE /W4 /UNDEBUG)
  target_compile_options(dyx_h6c_static_gate PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(dyx_h6c_selection_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  target_compile_options(dyx_h6c_static_gate PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
add_test(NAME h6c_selection_contract COMMAND dyx_h6c_selection_tests)
add_test(NAME h6c_no_actuation_source_gate COMMAND dyx_h6c_static_gate
  "${CMAKE_CURRENT_SOURCE_DIR}/../../src/dyx_mission_cpp17_preview/src/mission_preview_node.cpp")
